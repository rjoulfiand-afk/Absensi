1. Eksekusi Database (phpMyAdmin)
Buat database baru bernama db_waktu, lalu jalankan query SQL ini untuk membuat tabel akun (users) dan tabel riwayat waktu:

SQL
CREATE TABLE users (
    id_user INT AUTO_INCREMENT PRIMARY KEY,
    nama VARCHAR(100) NOT NULL,
    username VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL
);

CREATE TABLE log_waktu (
    id_log INT AUTO_INCREMENT PRIMARY KEY,
    id_user INT NOT NULL,
    nama_kegiatan VARCHAR(100) NOT NULL,
    waktu_mulai DATETIME NOT NULL,
    waktu_selesai DATETIME NOT NULL,
    total_durasi VARCHAR(50) NOT NULL
);

3. File Models (Logika Database)
A. models/UserModel.php (Untuk Login & Register)

PHP
<?php
class UserModel {
    private $db;
    public function __construct($koneksi) { $this->db = $koneksi; }

    public function cekUsername($username) {
        $username = mysqli_real_escape_string($this->db, $username);
        return $this->db->query("SELECT * FROM users WHERE username = '$username'")->num_rows > 0;
    }

    public function register($nama, $username, $password) {
        $nama = mysqli_real_escape_string($this->db, $nama);
        $username = mysqli_real_escape_string($this->db, $username);
        // Enkripsi password standar pakai MD5 (bisa diupgrade ke password_hash)
        $password = md5($password); 
        return $this->db->query("INSERT INTO users (nama, username, password) VALUES ('$nama', '$username', '$password')");
    }

    public function login($username, $password) {
        $username = mysqli_real_escape_string($this->db, $username);
        $password = md5($password);
        $hasil = $this->db->query("SELECT * FROM users WHERE username = '$username' AND password = '$password'");
        return $hasil->fetch_assoc();
    }
}
?>
B. models/WaktuModel.php (Untuk Nyimpan Data Perhitungan)

PHP
<?php
class WaktuModel {
    private $db;
    public function __construct($koneksi) { $this->db = $koneksi; }

    public function simpanLog($id_user, $kegiatan, $mulai, $selesai, $durasi) {
        $kegiatan = mysqli_real_escape_string($this->db, $kegiatan);
        return $this->db->query("INSERT INTO log_waktu (id_user, nama_kegiatan, waktu_mulai, waktu_selesai, total_durasi) 
                                 VALUES ($id_user, '$kegiatan', '$mulai', '$selesai', '$durasi')");
    }

    public function getLogByUser($id_user) {
        return $this->db->query("SELECT * FROM log_waktu WHERE id_user = $id_user ORDER BY id_log DESC");
    }
}
?>
4. File Controllers (Pengatur Alur & Logika)
A. controllers/AuthController.php

PHP
<?php
require_once 'models/UserModel.php';

class AuthController {
    private $model;
    public function __construct($koneksi) { $this->model = new UserModel($koneksi); }

    public function prosesLogin() {
        if ($_SERVER['REQUEST_METHOD'] == 'POST') {
            $user = $this->model->login($_POST['username'], $_POST['password']);
            if ($user) {
                $_SESSION['id_user'] = $user['id_user'];
                $_SESSION['nama'] = $user['nama'];
                header("Location: index.php?page=dashboard&alert=login_sukses");
                exit;
            }
            header("Location: index.php?alert=login_gagal"); exit;
        }
        require_once 'views/login.php';
    }

    public function prosesRegister() {
        if ($_SERVER['REQUEST_METHOD'] == 'POST') {
            if ($this->model->cekUsername($_POST['username'])) {
                header("Location: index.php?page=register&alert=username_ada"); exit;
            }
            $this->model->register($_POST['nama'], $_POST['username'], $_POST['password']);
            header("Location: index.php?alert=register_sukses"); exit;
        }
        require_once 'views/register.php';
    }

    public function logout() {
        session_destroy();
        header("Location: index.php?alert=logout"); exit;
    }
}
?>
B. controllers/WaktuController.php (Inti Logika Penghitungan Waktu)

PHP
<?php
require_once 'models/WaktuModel.php';

class WaktuController {
    private $model;
    public function __construct($koneksi) { $this->model = new WaktuModel($koneksi); }

    public function index() {
        $id_user = $_SESSION['id_user'];
        $data_log = $this->model->getLogByUser($id_user);
        require_once 'views/dashboard.php';
    }

    // LOGIKA PENGHITUNGAN WAKTU ADA DI SINI
    public function hitungDurasi() {
        if ($_SERVER['REQUEST_METHOD'] == 'POST') {
            $kegiatan = $_POST['nama_kegiatan'];
            $waktu_mulai = $_POST['waktu_mulai']; // Format dari form datetime-local
            $waktu_selesai = $_POST['waktu_selesai'];

            // 1. Ubah string waktu menjadi Objek DateTime PHP
            $obj_mulai = new DateTime($waktu_mulai);
            $obj_selesai = new DateTime($waktu_selesai);

            // 2. Validasi: Waktu selesai tidak boleh lebih dulu dari waktu mulai
            if ($obj_selesai < $obj_mulai) {
                header("Location: index.php?page=dashboard&alert=waktu_error");
                exit;
            }

            // 3. Hitung selisih (difference) antara dua waktu
            $selisih = $obj_mulai->diff($obj_selesai);

            // 4. Format hasil logika menjadi teks yang mudah dibaca (Misal: 2 Jam 30 Menit)
            // $selisih->d (hari), $selisih->h (jam), $selisih->i (menit)
            $durasi_teks = "";
            if ($selisih->d > 0) $durasi_teks .= $selisih->d . " Hari ";
            if ($selisih->h > 0) $durasi_teks .= $selisih->h . " Jam ";
            $durasi_teks .= $selisih->i . " Menit";

            // 5. Simpan ke Database
            $this->model->simpanLog($_SESSION['id_user'], $kegiatan, $waktu_mulai, $waktu_selesai, $durasi_teks);

            header("Location: index.php?page=dashboard&alert=hitung_sukses");
            exit;
        }
    }
}
?>

6. Router Pengendali (index.php)
Taruh file ini di folder paling luar (project_waktu/index.php).

PHP
<?php
// Wajib ditaruh paling atas biar session nggak bentrok
session_start();

require_once 'config/database.php';
require_once 'controllers/AuthController.php';
require_once 'controllers/WaktuController.php';

$auth = new AuthController($koneksi);
$waktu = new WaktuController($koneksi);

$page = isset($_GET['page']) ? $_GET['page'] : '';

switch ($page) {
    // RUTE AUTENTIKASI
    case 'register':
        $auth->prosesRegister();
        break;
    case 'logout':
        $auth->logout();
        break;

    // RUTE DASHBOARD & WAKTU (Harus Login Dulu)
    case 'dashboard':
        if (!isset($_SESSION['id_user'])) { header("Location: index.php"); exit; }
        $waktu->index();
        break;
    case 'hitung_waktu':
        if (!isset($_SESSION['id_user'])) { header("Location: index.php"); exit; }
        $waktu->hitungDurasi();
        break;

    // DEFAULT (Halaman Login)
    default:
        if (isset($_SESSION['id_user'])) { header("Location: index.php?page=dashboard"); exit; }
        $auth->prosesLogin();
        break;
}
?>
