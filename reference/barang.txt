BAGIAN 1: PONDASI DATABASE (phpMyAdmin)
====================================================================
Buat database baru bernama: `db_vendor`
Lalu jalankan SQL ini untuk membuat tabel users dan tabel data_vendor:

CREATE TABLE users (
    id_user INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    nama_lengkap VARCHAR(100) NOT NULL
);

-- Bikin akun admin default (Password: 12345)
INSERT INTO users (username, password, nama_lengkap) 
VALUES ('admin', '12345', 'Administrator');

CREATE TABLE data_vendor (
    id_vendor INT AUTO_INCREMENT PRIMARY KEY,
    nama_barang VARCHAR(100) NOT NULL,
    nama_vendor VARCHAR(100) NOT NULL,
    alamat TEXT NOT NULL,
    tanggal_masuk TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

--- A. models/UserModel.php (Khusus ngurus Login) ---
<?php
class UserModel {
    private $db;
    public function __construct($koneksi) { $this->db =$koneksi; }

    public function cekLogin($username, $password) {$username = mysqli_real_escape_string($this->db,$username);
        $password = mysqli_real_escape_string($this->db, $password);$query = "SELECT * FROM users WHERE username = '$username' AND password = '$password'";
        $hasil = $this->db->query($query);
        
        if($hasil->num_rows > 0) {
            return $hasil->fetch_assoc(); // Bawa balik data usernya
        }
        return false;
    }
}
?>

--- B. models/VendorModel.php (Khusus ngurus Data Vendor) ---
<?php
class VendorModel {
    private $db;
    public function __construct($koneksi) { $this->db =$koneksi; }

    public function getSemuaVendor() {
        return $this->db->query("SELECT * FROM data_vendor ORDER BY id_vendor DESC");
    }

    public function tambahVendor($barang,$vendor, $alamat) {$barang = mysqli_real_escape_string($this->db,$barang);
        $vendor = mysqli_real_escape_string($this->db, $vendor);$alamat = mysqli_real_escape_string($this->db,$alamat);
        
        return $this->db->query("INSERT INTO data_vendor (nama_barang, nama_vendor, alamat) VALUES ('$barang', '$vendor', '$alamat')");
    }

    public function hapusVendor($id) {
        $id = (int)$id;
        return $this->db->query("DELETE FROM data_vendor WHERE id_vendor = $id");
    }
}
?>

BAGIAN 4: CONTROLLERS (Pengatur Lalu Lintas)

--- A. controllers/AuthController.php ---
<?php
require_once 'models/UserModel.php';

class AuthController {
    private $model;
    public function __construct($koneksi) { 
        $this->model = new UserModel($koneksi); 
        if (session_status() === PHP_SESSION_NONE) { session_start(); }
    }

    public function login() {
        // Jika sudah login, tendang ke dashboard
        if(isset($_SESSION['is_login'])) { header("Location: index.php?page=dashboard"); exit; }

        if ($_SERVER['REQUEST_METHOD'] == 'POST') {
            $user =$_POST['username'];
            $pass =$_POST['password'];
            $cek =$this->model->cekLogin($user,$pass);

            if ($cek) {$_SESSION['is_login'] = true;
                $_SESSION['nama_lengkap'] =$cek['nama_lengkap'];
                header("Location: index.php?page=dashboard");
                exit;
            } else {
                header("Location: index.php?error=1");
                exit;
            }
        }
        require_once 'views/login.php';
    }

    public function logout() {
        session_destroy();
        header("Location: index.php");
        exit;
    }
}
?>

--- B. controllers/VendorController.php ---
<?php
require_once 'models/VendorModel.php';

class VendorController {
    private $model;
    public function __construct($koneksi) { $this->model = new VendorModel($koneksi); }

    public function index() {
        $data_vendor =$this->model->getSemuaVendor();
        require_once 'views/vendor_tampil.php';
    }

    public function tambah() {
        if ($_SERVER['REQUEST_METHOD'] == 'POST') {
            $barang =$_POST['nama_barang'];
            $vendor =$_POST['nama_vendor'];
            $alamat =$_POST['alamat'];

            if ($this->model->tambahVendor($barang, $vendor,$alamat)) {
                header("Location: index.php?page=vendor&pesan=sukses_tambah");
                exit;
            }
        }
        require_once 'views/vendor_tambah.php';
    }

    public function hapus() {
        if (isset($_GET['id'])) {
            $this->model->hapusVendor($_GET['id']);
        }
        header("Location: index.php?page=vendor&pesan=sukses_hapus");
        exit;
    }
}
?>

BAGIAN 6: ROUTER UTAMA (index.php)
====================================================================
Letakkan file ini di paling luar folder project lu.
Ini adalah pusat kendali yang menyambungkan semua file di atas.

<?php
require_once 'config/database.php';
require_once 'controllers/AuthController.php';
require_once 'controllers/VendorController.php';

$auth = new AuthController($koneksi);
$vendor = new VendorController($koneksi);

$page = isset($_GET['page']) ?$_GET['page'] : '';

switch ($page) {
    case 'dashboard':
        require_once 'views/dashboard.php';
        break;
        
    case 'vendor':
        $vendor->index();
        break;
        
    case 'tambah_vendor':
        $vendor->tambah();
        break;
        
    case 'hapus_vendor':
        $vendor->hapus();
        break;
        
    case 'logout':
        $auth->logout();
        break;

    default:
        // Kalau URL kosong atau nggak valid, arahkan ke login
        $auth->login();
        break;
}
?>
