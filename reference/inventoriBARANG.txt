1. ARSITEKTUR ALUR & LOGIKA STOK ANTI-MINUS
====================================================================================================
A. Struktur Multi-Stok:
   Tabel barang memiliki 2 kolom stok: `stok_total` (jumlah aset fisik sekolah) dan `stok_tersedia` 
   (stok yang saat ini ada di gudang dan siap dipinjam). 

B. Siklus Transaksi Peminjaman (Pintu Keluar):
   [Siswa/Guru Pinjam 2 Unit] -> [Start Transaction] -> [Kunci Baris Barang `FOR UPDATE`] 
   -> [Cek: Apakah stok_tersedia >= 2?] 
      -> Jika YA: Insert data peminjaman -> Update stok_tersedia = stok_tersedia - 2 -> Commit.
      -> Jika TIDAK: Rollback -> Tampilkan Error "Stok Tidak Cukup".

C. Siklus Pengembalian (Pintu Masuk):
   [Klik Tombol Kembalikan] -> [Start Transaction] -> [Cek Status Peminjaman]
   -> Jika status masih 'Dipinjam': Update status jadi 'Dikembalikan' & catat jam pengembalian
   -> Update stok_tersedia = stok_tersedia + 2 -> Commit.

class BarangModel {
    private $db;
    public function __construct($koneksi) { $this->db =$koneksi; }

    public function getAllBarang() {
        return $this->db->query("SELECT * FROM barang ORDER BY nama_barang ASC");
    }

    public function getBarangTersedia() {
        return $this->db->query("SELECT * FROM barang WHERE stok_tersedia > 0 ORDER BY nama_barang ASC");
    }

    public function tambahBarang($kode,$nama, $kategori,$stok) {
        $stmt =$this->db->prepare("INSERT INTO barang (kode_barang, nama_barang, kategori, stok_total, stok_tersedia) VALUES (?, ?, ?, ?, ?)");
        $stmt->bind_param("sssii", $kode, $nama,$kategori, $stok,$stok);
        return $stmt->execute();
    }
}

--- C. models/TransaksiModel.php (SISTEM KUNCI STOK ACID) ---
<?php
class TransaksiModel {
    private $db;
    public function __construct($koneksi) { $this->db =$koneksi; }

    public function getRiwayat() {
        $query = "SELECT p.*, b.nama_barang, b.kode_barang 
                  FROM peminjaman p 
                  JOIN barang b ON p.id_barang = b.id_barang 
                  ORDER BY p.tgl_pinjam DESC";
        return $this->db->query($query);
    }

    // LOGIKA PEMINJAMAN (MENGURANGI STOK AMAN)
    public function pinjamBarang($id_barang,$peminjam, $kelas,$jumlah, $ket) {$this->db->begin_transaction();
        try {
            // 1. Kunci baris barang agar tidak dibaca transaksi lain bersamaan
            $stmtCek =$this->db->prepare("SELECT stok_tersedia FROM barang WHERE id_barang = ? FOR UPDATE");
            $stmtCek->bind_param("i", $id_barang);$stmtCek->execute();
            $hasil =$stmtCek->get_result()->fetch_assoc();

            // 2. Validasi Ketersediaan Stok
            if (!$hasil \vert{}\vert{}$hasil['stok_tersedia'] < $jumlah) {$this->db->rollback();
                return 'STOK_KURANG';
            }

            // 3. Generate Kode Transaksi Unik (Contoh: TRX-20260922-1234)
            $kode_trx = "TRX-" . date('Ymd') . "-" . rand(1000, 9999);
            $waktu = date('Y-m-d H:i:s');

            // 4. Catat ke Tabel Peminjaman
            $stmtPinjam =$this->db->prepare("INSERT INTO peminjaman (kode_transaksi, id_barang, peminjam, kelas_instansi, jumlah_pinjam, tgl_pinjam, status, keterangan) VALUES (?, ?, ?, ?, ?, ?, 'Dipinjam', ?)");
            $stmtPinjam->bind_param("sisssis", $kode_trx,$id_barang, $peminjam,$kelas, $jumlah,$waktu, $ket);$stmtPinjam->execute();

            // 5. Potong Stok Tersedia
            $stmtStok =$this->db->prepare("UPDATE barang SET stok_tersedia = stok_tersedia - ? WHERE id_barang = ?");
            $stmtStok->bind_param("ii", $jumlah, $id_barang);$stmtStok->execute();

            $this->db->commit();
            return 'SUKSES';

        } catch (Exception $e) {$this->db->rollback();
            return 'ERROR_SYSTEM';
        }
    }

    // LOGIKA PENGEMBALIAN (MENAMBAH STOK KEMBALI)
    public function kembalikanBarang($id_pinjam) {$this->db->begin_transaction();
        try {
            // 1. Kunci data peminjaman
            $stmtCek =$this->db->prepare("SELECT id_barang, jumlah_pinjam, status FROM peminjaman WHERE id_pinjam = ? FOR UPDATE");
            $stmtCek->bind_param("i", $id_pinjam);$stmtCek->execute();
            $trx =$stmtCek->get_result()->fetch_assoc();

            // Mencegah klik ganda tombol kembali
            if (!$trx || $trx['status'] === 'Dikembalikan') {$this->db->rollback();
                return false; 
            }

            $waktu_kembali = date('Y-m-d H:i:s');

            // 2. Ubah status dan catat waktu kembali
            $stmtUpdateTrx =$this->db->prepare("UPDATE peminjaman SET status = 'Dikembalikan', tgl_kembali = ? WHERE id_pinjam = ?");
            $stmtUpdateTrx->bind_param("si", $waktu_kembali, $id_pinjam);$stmtUpdateTrx->execute();

            // 3. Kembalikan stok ke gudang
            $stmtRestok =$this->db->prepare("UPDATE barang SET stok_tersedia = stok_tersedia + ? WHERE id_barang = ?");
            $stmtRestok->bind_param("ii", $trx['jumlah_pinjam'], $trx['id_barang']);$stmtRestok->execute();

            $this->db->commit();
            return true;
        } catch (Exception $e) {$this->db->rollback();
            return false;
        }
    }
}
?>

--- controllers/TransaksiController.php ---
<?php
require_once 'models/TransaksiModel.php';
require_once 'models/BarangModel.php';

class TransaksiController {
    private $trxModel;
    private $brgModel;

    public function __construct($koneksi) {
        $this->trxModel = new TransaksiModel($koneksi);
        $this->brgModel = new BarangModel($koneksi);
        if (session_status() === PHP_SESSION_NONE) session_start();
    }

    public function index() {
        $riwayat =$this->trxModel->getRiwayat();
        require_once 'views/log_peminjaman.php';
    }

    public function form_pinjam() {
        if ($_SERVER['REQUEST_METHOD'] === 'POST') {
            $id_barang = (int)$_POST['id_barang'];
            $peminjam =$_POST['peminjam'];
            $kelas =$_POST['kelas'];
            $jumlah = (int)$_POST['jumlah'];
            $ket =$_POST['keterangan'];

            if ($jumlah <= 0) {
                header("Location: index.php?page=transaksi&pesan=jumlah_invalid"); exit;
            }

            $proses = $this->trxModel->pinjamBarang($id_barang, $peminjam,$kelas, $jumlah,$ket);

            if ($proses === 'SUKSES') {                 header("Location: index.php?page=transaksi&pesan=pinjam_sukses"); exit;             } elseif ($proses === 'STOK_KURANG') {
                header("Location: index.php?page=transaksi&pesan=stok_limit"); exit;
            } else {
                header("Location: index.php?page=transaksi&pesan=error_server"); exit;
            }
        }
        
        $barang_siap =$this->brgModel->getBarangTersedia();
        require_once 'views/form_pinjam.php';
    }

    public function aksi_kembali() {
        if (isset($_GET['id'])) {$sukses = $this->trxModel->kembalikanBarang($_GET['id']);
            if ($sukses) {
                header("Location: index.php?page=transaksi&pesan=kembali_sukses"); exit;
            } else {
                header("Location: index.php?page=transaksi&pesan=gagal_kembali"); exit;
            }
        }
    }
}
?>
