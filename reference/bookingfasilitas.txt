CREATE TABLE tabel_parkir (
    id_parkir INT AUTO_INCREMENT PRIMARY KEY,
    nomor_plat VARCHAR(20) NOT NULL,
    jenis_kendaraan ENUM('Roda 2', 'Roda 4') NOT NULL,
    waktu_masuk DATETIME NOT NULL,
    waktu_keluar DATETIME NULL,
    status ENUM('Belum', 'Selesai') DEFAULT 'Belum',
    total_bayar INT(11) DEFAULT 0
);

class AdminModel {
    private $db;
    public function __construct($koneksi) { $this->db = $koneksi; }
    public function login($username, $password) {
        $username = mysqli_real_escape_string($this->db, $username);
        // Login standar (Plain text) sesuai kebutuhan ujian cepat
        $hasil = $this->db->query("SELECT * FROM admin WHERE username='$username' AND password='$password'");
        return $hasil->fetch_assoc();
    }

--- models/ParkirModel.php ---
<?php
class ParkirModel {
    private $db;
    public function __construct($koneksi) { $this->db = $koneksi; }

    // Dapatkan data kendaraan yang sedang parkir
    public function getParkirAktif() {
        return $this->db->query("SELECT * FROM tabel_parkir WHERE status = 'Belum' ORDER BY waktu_masuk DESC");
    }

    // Dapatkan semua data (Untuk Laporan)
    public function getLaporanHarian($tanggal) {
        // Ambil data yang waktu masuknya di hari yang dipilih
        return $this->db->query("SELECT * FROM tabel_parkir WHERE DATE(waktu_masuk) = '$tanggal' ORDER BY waktu_masuk DESC");
    }

    // Hitung total pendapatan hari ini
    public function getPendapatanHariIni($tanggal) {
        $hasil = $this->db->query("SELECT SUM(total_bayar) as total FROM tabel_parkir WHERE DATE(waktu_keluar) = '$tanggal' AND status = 'Selesai'");
        $data = $hasil->fetch_assoc();
        return $data['total'] ? $data['total'] : 0;
    }

    // PARKIR MASUK (CREATE)
    public function catatMasuk($plat, $jenis, $waktu_masuk) {
        $plat = strtoupper(mysqli_real_escape_string($this->db, $plat));
        return $this->db->query("INSERT INTO tabel_parkir (nomor_plat, jenis_kendaraan, waktu_masuk, status) 
                                 VALUES ('$plat', '$jenis', '$waktu_masuk', 'Belum')");
    }

    // PENCARIAN PARKIR KELUAR (READ berdasarkan PLAT)
    public function cariKendaraanKeluar($plat) {
        $plat = strtoupper(mysqli_real_escape_string($this->db, $plat));
        // Cari HANYA yang statusnya 'Belum' (masih di dalam)
        $hasil = $this->db->query("SELECT * FROM tabel_parkir WHERE nomor_plat LIKE '%$plat%' AND status = 'Belum'");
        return $hasil->fetch_assoc(); 
    }

    // EKSEKUSI PARKIR KELUAR & SIMPAN BAYAR (UPDATE)
    public function prosesKeluar($id_parkir, $waktu_keluar, $total_bayar) {
        $id = (int)$id_parkir;
        $bayar = (int)$total_bayar;
        return $this->db->query("UPDATE tabel_parkir 
                                 SET waktu_keluar = '$waktu_keluar', status = 'Selesai', total_bayar = $bayar 
                                 WHERE id_parkir = $id");
    }
}
?>

            header("Location: index.php?pesan=gagal"); exit;

        require_once 'views/login.php';
    }
    public function logout() { session_destroy(); header("Location: index.php"); exit; }
}

--- controllers/ParkirController.php ---
<?php
require_once 'models/ParkirModel.php';

class ParkirController {
    private $model;
    public function __construct($koneksi) { $this->model = new ParkirModel($koneksi); }

    // 1. DASHBOARD UTAMA
    public function dashboard() {
        $tgl_hari_ini = date('Y-m-d');$kendaraan_aktif = $this->model->getParkirAktif();$pendapatan = $this->model->getPendapatanHariIni($tgl_hari_ini);
        require_once 'views/dashboard.php';
    }

    // 2. PARKIR MASUK
    public function parkir_masuk() {
        if($_SERVER['REQUEST_METHOD'] == 'POST') {$plat = $_POST['nomor_plat'];$jenis = $_POST['jenis_kendaraan'];$waktu_masuk = date('Y-m-d H:i:s'); // Rekam jam persis pas tombol diklik

            $this->model->catatMasuk($plat, $jenis,$waktu_masuk);
            header("Location: index.php?page=parkir_masuk&pesan=sukses_masuk"); exit;
        }
        require_once 'views/parkir_masuk.php';
    }

    // 3. PARKIR KELUAR (Pencarian & Hitung Bayar)
    public function parkir_keluar() {
        $kendaraan = null;
        $tagihan = 0;
        $waktu_keluar = date('Y-m-d H:i:s');$durasi_jam = 0;

        // Jika Admin nyari Plat Nomor
        if(isset($_GET['cari_plat'])) {$kendaraan = $this->model->cariKendaraanKeluar($_GET['cari_plat']);
            
            // LOGIKA MATEMATIKA PARKIR (KILLER FEATURE!)
            if($kendaraan) {
                // Konversi string waktu ke Unix Timestamp (detik)
                $masuk = strtotime($kendaraan['waktu_masuk']);
                $keluar = strtotime($waktu_keluar);
                
                // Hitung selisih dalam jam (dibulatkan ke atas)
                $selisih_detik = $keluar -$masuk;
                $durasi_jam = ceil($selisih_detik / 3600);
                if($durasi_jam < 1)$durasi_jam = 1; // Minimal hitungan 1 jam
                
                // Rumus Uang
                $tarif_dasar = ($kendaraan['jenis_kendaraan'] == 'Roda 2') ? 2000 : 5000;
                $tambahan = 0;
                
                // Jika lebih dari 2 jam, kena denda progresif 1k/jam
                if($durasi_jam > 2) {$tambahan = ($durasi_jam - 2) * 1000;
                }$tagihan = $tarif_dasar +$tambahan;
            }
        }

        // Jika Admin menekan tombol "Proses Pembayaran"
        if($_SERVER['REQUEST_METHOD'] == 'POST') {
            $this->model->prosesKeluar($_POST['id_parkir'], $_POST['waktu_keluar'],$_POST['total_bayar']);
            header("Location: index.php?page=parkir_keluar&pesan=sukses_keluar"); exit;
        }

        require_once 'views/parkir_keluar.php';
    }

    // 4. LAPORAN & CETAK PDF
    public function laporan() {
        // Default nampilin hari ini. Kalau milih tanggal di form, pakai tanggal itu.
        $tanggal = isset($_GET['tanggal']) ?$_GET['tanggal'] : date('Y-m-d');
        $data_laporan =$this->model->getLaporanHarian($tanggal);$total_pendapatan = $this->model->getPendapatanHariIni($tanggal);
        
        require_once 'views/laporan.php';
    }

    public function cetak_pdf() {
        $tanggal =$_GET['tanggal'];
        $data_laporan =$this->model->getLaporanHarian($tanggal);$total_pendapatan = $this->model->getPendapatanHariIni($tanggal);
        require_once 'views/cetak_pdf.php';
    }
}

Bedah Fitur Khusus:
Fitur Cari & Hitung (Parkir Keluar): Saat admin ngetik plat nomor, sistem ngebaca URL GET (misal: index.php?page=parkir_keluar&cari_plat=L1234AB). Di controller, selisih waktu (waktu_keluar - waktu_masuk) diubah jadi jam menggunakan rumus ceil(detik / 3600). Jika > 2, denda progresif 1000 dikalikan kelebihan jamnya.

Cetak PDF Smart Hack (window.print()): Di ujian SMK, menginstall TCPDF/DomPDF itu makan waktu dan rawan error server. Pakai trik ini, pas admin klik "Unduh PDF", tab baru kebuka dan langsung melempar perintah ke Browser untuk nge-print. Admin tinggal pilih "Save as PDF" / "Simpan sebagai PDF" di jendela yang muncul.

Validasi Anti-Keluar Nyasar (Alert Logout): Gw udah tambahin trigger onclick="konfirmasiKeluar()" di navbar. Kalau lu ngeklik tombol "Keluar APK", sistem gak langsung mati, tapi ngeluarin notif SweetAlert untuk Make Sure. Kalau di "OK", JS bakal nge-redirect (lempar URL) ke index.php?page=logout.
