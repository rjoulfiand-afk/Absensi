CREATE TABLE users (

    nama_lengkap VARCHAR(100) NOT NULL,
    username VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL
);
INSERT INTO users (nama_lengkap, username, password) VALUES ('Admin Fasilitas', 'admin', '12345');

CREATE TABLE jadwal_booking (
    id_booking INT AUTO_INCREMENT PRIMARY KEY,
    peminjam VARCHAR(100) NOT NULL,
    fasilitas VARCHAR(100) NOT NULL,
    tgl_booking DATE NOT NULL,
    jam_mulai TIME NOT NULL,
    jam_selesai TIME NOT NULL,
    keperluan TEXT NOT NULL,
    status ENUM('Disetujui', 'Dibatalkan') DEFAULT 'Disetujui'
);

--- C. models/BookingModel.php ---
<?php
class BookingModel {
    private $db;
    public function __construct($koneksi) { $this->db =$koneksi; }

    // RUMUS AJAIB ANTI-BENTROK WAKTU
    // Parameter $id_ignore dipakai saat fitur Edit, agar sistem tidak menganggap bentrok dengan data dirinya sendiri.
    public function cekBentrok($fasilitas, $tgl,$mulai, $selesai,$id_ignore = 0) {
        $fasilitas = mysqli_real_escape_string($this->db, $fasilitas);$query = "SELECT * FROM jadwal_booking 
                  WHERE fasilitas = '$fasilitas' 
                  AND tgl_booking = '$tgl' 
                  AND (jam_mulai < '$selesai' AND jam_selesai > '$mulai')
                  AND id_booking != $id_ignore 
                  AND status = 'Disetujui'";
                  
        $hasil = $this->db->query($query);
        return $hasil->num_rows > 0; // Return TRUE jika jadwal bentrok!
    }

    public function getAll() {
        return $this->db->query("SELECT * FROM jadwal_booking ORDER BY tgl_booking DESC, jam_mulai ASC");
    }

    public function getById($id) {
        return $this->db->query("SELECT * FROM jadwal_booking WHERE id_booking = ".(int)$id)->fetch_assoc();
    }

    public function tambah($peminjam,$fasilitas, $tgl,$mulai, $selesai,$keperluan) {
        $peminjam = mysqli_real_escape_string($this->db, $peminjam);$keperluan = mysqli_real_escape_string($this->db,$keperluan);
        return $this->db->query("INSERT INTO jadwal_booking (peminjam, fasilitas, tgl_booking, jam_mulai, jam_selesai, keperluan) 
                                 VALUES ('$peminjam', '$fasilitas', '$tgl', '$mulai', '$selesai', '$keperluan')");
    }

    public function update($id,$peminjam, $fasilitas,$tgl, $mulai,$selesai, $keperluan,$status) {
        $id = (int)$id;
        $peminjam = mysqli_real_escape_string($this->db, $peminjam);$keperluan = mysqli_real_escape_string($this->db,$keperluan);
        return $this->db->query("UPDATE jadwal_booking SET peminjam='$peminjam', fasilitas='$fasilitas', 
                                 tgl_booking='$tgl', jam_mulai='$mulai', jam_selesai='$selesai', 
                                 keperluan='$keperluan', status='$status' WHERE id_booking=$id");
    }

    public function hapus($id) {
        return $this->db->query("DELETE FROM jadwal_booking WHERE id_booking = ".(int)$id);
    }
}
?>

--- A. controllers/AuthController.php ---
<?php
require_once 'models/UserModel.php';
class AuthController {
    private $model;
    public function __construct($koneksi) { 
        $this->model = new UserModel($koneksi); 
        if(session_status() === PHP_SESSION_NONE) session_start();
    }
    public function login() {
        if(isset($_SESSION['id_user'])) { header("Location: index.php?page=dashboard"); exit; }
        if($_SERVER['REQUEST_METHOD'] == 'POST') {
            $user =$this->model->login($_POST['username'],$_POST['password']);
            if($user) {
                $_SESSION['id_user'] =$user['id_user'];
                $_SESSION['nama_lengkap'] =$user['nama_lengkap'];
                header("Location: index.php?page=dashboard&alert=welcome"); exit;
            }
            header("Location: index.php?alert=gagal"); exit;
        }
        require_once 'views/login.php';
    }
    public function logout() { session_destroy(); header("Location: index.php"); exit; }
}
?>

--- B. controllers/BookingController.php ---
<?php
require_once 'models/BookingModel.php';
class BookingController {
    private $model;
    public function __construct($koneksi) { $this->model = new BookingModel($koneksi); }

    public function index() {
        $data_booking =$this->model->getAll();
        require_once 'views/booking_tampil.php';
    }

    public function form_tambah() {
        if($_SERVER['REQUEST_METHOD'] == 'POST') {
            $fasilitas =$_POST['fasilitas'];
            $tgl =$_POST['tgl_booking'];
            $mulai =$_POST['jam_mulai'];
            $selesai =$_POST['jam_selesai'];

            // 1. Validasi Logika Waktu Awal
            if($selesai <=$mulai) {
                header("Location: index.php?page=tambah_booking&pesan=waktu_salah"); exit;
            }

            // 2. Validasi Bentrok Jadwal di Database
            if($this->model->cekBentrok($fasilitas,$tgl, $mulai,$selesai)) {
                header("Location: index.php?page=tambah_booking&pesan=bentrok"); exit;
            }

            // 3. Jika aman, eksekusi Simpan
            $this->model->tambah($_POST['peminjam'],$fasilitas, $tgl,$mulai, $selesai,$_POST['keperluan']);
            header("Location: index.php?page=booking&pesan=tambah"); exit;
        }
        require_once 'views/booking_tambah.php';
    }

    public function form_edit() {
        if($_SERVER['REQUEST_METHOD'] == 'POST') {
            $id =$_POST['id_booking'];
            $fasilitas =$_POST['fasilitas'];
            $tgl =$_POST['tgl_booking'];
            $mulai =$_POST['jam_mulai'];
            $selesai =$_POST['jam_selesai'];
            $status =$_POST['status'];

            // 1. Validasi Logika Waktu Awal
            if($selesai <=$mulai) {
                header("Location: index.php?page=edit_booking&id=$id&pesan=waktu_salah"); exit;
            }

            // 2. Validasi Bentrok (dengan $id_ignore agar bisa edit dirinya sendiri)
            // Mengecek jika dia ganti jadwal, apakah bentrok sama jadwal orang lain? Tapi kalau Dibatalkan, biarin aja.
            if($status == 'Disetujui' && $this->model->cekBentrok($fasilitas, $tgl,$mulai, $selesai,$id)) {
                header("Location: index.php?page=edit_booking&id=$id&pesan=bentrok"); exit;
            }

            $this->model->update($id,$_POST['peminjam'], $fasilitas,$tgl, $mulai,$selesai, $_POST['keperluan'],$status);
            header("Location: index.php?page=booking&pesan=edit"); exit;
        } else {
            $data_edit = $this->model->getById($_GET['id']);
            require_once 'views/booking_edit.php';
        }
    }

    public function aksi_hapus() {
        if(isset($_GET['id'])) $this->model->hapus($_GET['id']);
        header("Location: index.php?page=booking&pesan=hapus"); exit;
    }
}
?>
