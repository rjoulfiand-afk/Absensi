====================================================================================================
1. ARSITEKTUR ALUR & PROTOKOL KEAMANAN (ANTI-FRAUD)
====================================================================================================
A. Autentikasi Ganda (Role Separation):
   - Admin/Panitia: Menggunakan pasangan Username & Password terenkripsi Hash. Mendapatkan hak akses
     ke Dashboard Monitoring dan pengawasan hasil suara.
   - Pemilih (Siswa): Menggunakan Nomor Induk Siswa Nasional (NISN) dan Token Akses Rahasia 6 digit.
     Hanya dapat melihat halaman bilik pencoblosan (Ballot Box).

B. Siklus Transaksi Hak Suara (State Machine):
   [Siswa Login] -> [Validasi Token & Status Vote] -> [Masuk Bilik] 
                 -> [Kirim Suara via POST] -> [Database Transaction Start]
                 -> [Update Jumlah Suara Paslon] -> [Kunci Status Pemilih = 'Sudah'] 
                 -> [Commit Transaksi] -> [Regenerate Sesi / Destroy Sesi] -> [Keluar]

C. Penanganan Race Condition & Atomic Locks:
   Untuk mencegah anomali di mana 1 pemilih membuka 2 tab browser secara bersamaan dan menekan tombol
   coblos dalam milidetik yang sama, Model menggunakan `mysqli_begin_transaction()` serta validasi baris
   ganda: pemilih hanya dapat dieksekusi jika statusnya saat itu masih 'Belum'.

D. Live Data Polling (Asynchronous):
   Dashboard Admin tidak melakukan refresh halaman secara manual (`F5`). JavaScript (Fetch API) secara 
   berkala mengirimkan HTTP Request latar belakang setiap 2500ms ke Router (`?page=api_live_count`).
   Router mengembalikan data JSON teragregasi yang langsung diinjeksikan ke lebar elemen Progress Bar.

<?php
class AdminModel {
    private $db;

    public function __construct($koneksi) {
        $this->db = $koneksi;
    }

    public function verifikasiLogin($username, $password) {
        $stmt = $this->db->prepare("SELECT id_admin, nama_petugas, username, password_hash FROM users_admin WHERE username = ?");
        $stmt->bind_param("s", $username);
        $stmt->execute();
        $hasil = $stmt->get_result();

        if ($hasil->num_rows === 1) {
            $admin = $hasil->fetch_assoc();
            // Verifikasi Password Hash Bcrypt
            if (password_verify($password, $admin['password_hash'])) {
                return $admin;
            }
        }
        return false;
    }
}
?>

class PemilihModel {
    private $db;

    public function __construct($koneksi) {
        $this->db = $koneksi;
    }

    public function verifikasiPemilih($nisn, $token) {
        $stmt = $this->db->prepare("SELECT id_pemilih, nisn, nama_siswa, kelas, status_vote FROM pemilih WHERE nisn = ? AND token_akses = ?");
        $stmt->bind_param("ss", $nisn, $token);
        $stmt->execute();
        return $stmt->get_result()->fetch_assoc();
    }

    public function getPemilihById($id_pemilih) {
        $stmt = $this->db->prepare("SELECT id_pemilih, nisn, nama_siswa, kelas, status_vote FROM pemilih WHERE id_pemilih = ?");
        $stmt->bind_param("i", $id_pemilih);
        $stmt->execute();
        return $stmt->get_result()->fetch_assoc();
    }
}

--- C. models/KandidatModel.php ---
<?php
class KandidatModel {
    private $db;

    public function __construct($koneksi) {
        $this->db = $koneksi;
    }

    public function getSemuaKandidat() {
        return $this->db->query("SELECT * FROM kandidat ORDER BY no_urut ASC");
    }

    public function getKandidatById($id) {
        $stmt = $this->db->prepare("SELECT * FROM kandidat WHERE id_kandidat = ?");
        $stmt->bind_param("i", $id);
        $stmt->execute();
        return $stmt->get_result()->fetch_assoc();
    }

    // TRANSAKSI DATABASE KETAT (ACID): Mencegah Suara Dobel & Data Corrupt
    public function rekamPilihanSiswa($id_pemilih, $id_kandidat) {
        // 1. Mulai Transaksi
        $this->db->begin_transaction();

        try {
            // Validasi baris kunci: Pastikan siswa belum memilih
            $stmtCek = $this->db->prepare("SELECT status_vote FROM pemilih WHERE id_pemilih = ? FOR UPDATE");
            $stmtCek->bind_param("i", $id_pemilih);
            $stmtCek->execute();
            $hasil = $stmtCek->get_result()->fetch_assoc();

            if (!$hasil || $hasil['status_vote'] === 'Sudah') {
                $this->db->rollback();
                return false; // Ditolak karena manipulasi / sudah memilih
            }

            // 2. Tambah 1 suara pada Paslon yang bersangkutan
            $stmtSuara = $this->db->prepare("UPDATE kandidat SET total_suara = total_suara + 1 WHERE id_kandidat = ?");
            $stmtSuara->bind_param("i", $id_kandidat);
            $stmtSuara->execute();

            // 3. Kunci status siswa menjadi 'Sudah' dan pasang stempel waktu
            $waktu = date('Y-m-d H:i:s');
            $stmtKunci = $this->db->prepare("UPDATE pemilih SET status_vote = 'Sudah', waktu_memilih = ? WHERE id_pemilih = ?");
            $stmtKunci->bind_param("si", $waktu, $id_pemilih);
            $stmtKunci->execute();

            // 4. Komit seluruh transaksi ke storage
            $this->db->commit();
            return true;
        } catch (Exception $e) {
            $this->db->rollback();
            return false;
        }
    }

    // KALKULASI PERSENTASE REAL-TIME (Untuk konsumsi JSON Endpoint)
    public function getRekapSuaraRealtime() {
        // Ambil akumulasi suara masuk
        $qTotal = $this->db->query("SELECT SUM(total_suara) as grand_total FROM kandidat");
        $grand_total = (int)$qTotal->fetch_assoc()['grand_total'];

        // Ambil status partisipasi pemilih
        $qPartisipasi = $this->db->query("SELECT 
            COUNT(*) as total_pemilih,
            SUM(CASE WHEN status_vote = 'Sudah' THEN 1 ELSE 0 END) as sudah_memilih,
            SUM(CASE WHEN status_vote = 'Belum' THEN 1 ELSE 0 END) as belum_memilih
            FROM pemilih");
        $partisipasi = $qPartisipasi->fetch_assoc();

        $kandidatData = $this->db->query("SELECT id_kandidat, no_urut, CONCAT(nama_ketua, ' & ', nama_wakil) as paslon, total_suara, warna_tema FROM kandidat ORDER BY no_urut ASC");
        
        $listKandidat = [];
        while ($row = $kandidatData->fetch_assoc()) {
            $persen = ($grand_total > 0) ? round(($row['total_suara'] / $grand_total) * 100, 2) : 0;
            $listKandidat[] = [
                'id_kandidat' => (int)$row['id_kandidat'],
                'no_urut' => (int)$row['no_urut'],
                'paslon' => $row['paslon'],
                'suara' => (int)$row['total_suara'],
                'persentase' => $persen,
                'warna' => $row['warna_tema']
            ];
        }

        return [
            'grand_total_suara' => $grand_total,
            'partisipasi' => [
                'total_dpt' => (int)$partisipasi['total_pemilih'],
                'sudah_memilih' => (int)$partisipasi['sudah_memilih'],
                'belum_memilih' => (int)$partisipasi['belum_memilih'],
                'persentase_kehadiran' => ($partisipasi['total_pemilih'] > 0) ? round(($partisipasi['sudah_memilih'] / $partisipasi['total_pemilih']) * 100, 2) : 0
            ],
            'kandidat' => $listKandidat,
            'timestamp' => date('H:i:s')
        ];
    }
}

--- A. controllers/AuthController.php ---
<?php
require_once 'models/AdminModel.php';
require_once 'models/PemilihModel.php';

class AuthController {
    private $adminModel;
    private $pemilihModel;

    public function __construct($koneksi) {
        $this->adminModel = new AdminModel($koneksi);
        $this->pemilihModel = new PemilihModel($koneksi);
        if (session_status() === PHP_SESSION_NONE) session_start();
    }

    public function login() {
        // Redirection Guard jika sudah login
        if (isset($_SESSION['role'])) {
            if ($_SESSION['role'] === 'admin') { header("Location: index.php?page=dashboard_admin"); exit; }
            if ($_SESSION['role'] === 'pemilih') { header("Location: index.php?page=bilik_suara"); exit; }
        }

        if ($_SERVER['REQUEST_METHOD'] === 'POST') {
            $tipe_login = $_POST['tipe_login'] ?? '';

            // 1. Cabang Login Panitia (Admin)
            if ($tipe_login === 'admin') {
                $username = trim($_POST['username'] ?? '');
                $password = trim($_POST['password'] ?? '');
                $admin = $this->adminModel->verifikasiLogin($username, $password);

                if ($admin) {
                    session_regenerate_id(true);
                    $_SESSION['role'] = 'admin';
                    $_SESSION['user_id'] = $admin['id_admin'];
                    $_SESSION['nama'] = $admin['nama_petugas'];
                    header("Location: index.php?page=dashboard_admin&pesan=login_sukses");
                    exit;
                } else {
                    header("Location: index.php?pesan=auth_gagal"); exit;
                }
            }

            // 2. Cabang Login Siswa (Pemilih)
            if ($tipe_login === 'siswa') {
                $nisn = trim($_POST['nisn'] ?? '');
                $token = trim($_POST['token'] ?? '');
                $pemilih = $this->pemilihModel->verifikasiPemilih($nisn, $token);

                if ($pemilih) {
                    if ($pemilih['status_vote'] === 'Sudah') {
                        header("Location: index.php?pesan=sudah_memilih"); exit;
                    }

                    session_regenerate_id(true);
                    $_SESSION['role'] = 'pemilih';
                    $_SESSION['user_id'] = $pemilih['id_pemilih'];
                    $_SESSION['nama'] = $pemilih['nama_siswa'];
                    $_SESSION['kelas'] = $pemilih['kelas'];
                    header("Location: index.php?page=bilik_suara");
                    exit;
                } else {
                    header("Location: index.php?pesan=kredensial_siswa_salah"); exit;
                }
            }
        }

        require_once 'views/login.php';
    }

    public function logout() {
        $_SESSION = [];
        if (ini_get("session.use_cookies")) {
            $params = session_get_cookie_params();
            setcookie(session_name(), '', time() - 42000, $params["path"], $params["domain"], $params["secure"], $params["httponly"]);
        }
        session_destroy();
        header("Location: index.php?pesan=logout_sukses");
        exit;
    }
}
?>

--- B. controllers/BilikController.php ---
<?php
require_once 'models/KandidatModel.php';
require_once 'models/PemilihModel.php';

class BilikController {
    private $kandidatModel;
    private $pemilihModel;

    public function __construct($koneksi) {
        $this->kandidatModel = new KandidatModel($koneksi);
        $this->pemilihModel = new PemilihModel($koneksi);
        if (session_status() === PHP_SESSION_NONE) session_start();
        $this->guardPemilih();
    }

    private function guardPemilih() {
        if (!isset($_SESSION['role']) || $_SESSION['role'] !== 'pemilih') {
            header("Location: index.php?pesan=akses_ditolak");
            exit;
        }
    }

    public function index() {
        $id_pemilih = $_SESSION['user_id'];
        $data_siswa = $this->pemilihModel->getPemilihById($id_pemilih);

        // Jika data di database sudah status 'Sudah', paksa logout
        if ($data_siswa['status_vote'] === 'Sudah') {
            session_destroy();
            header("Location: index.php?pesan=sudah_memilih");
            exit;
        }

        $daftar_kandidat = $this->kandidatModel->getSemuaKandidat();
        require_once 'views/bilik_suara.php';
    }

    public function submitPilihan() {
        if ($_SERVER['REQUEST_METHOD'] === 'POST') {
            $id_pemilih = (int)$_SESSION['user_id'];
            $id_kandidat = (int)($_POST['id_kandidat'] ?? 0);

            if ($id_kandidat <= 0) {
                header("Location: index.php?page=bilik_suara&pesan=pilihan_tidak_valid");
                exit;
            }

            // Eksekusi Transaksi Database
            $sukses = $this->kandidatModel->rekamPilihanSiswa($id_pemilih, $id_kandidat);

            if ($sukses) {
                // Hancurkan Sesi Seketika (Zero Tolerance Re-Voting)
                session_destroy();
                header("Location: index.php?pesan=vote_berhasil");
                exit;
            } else {
                session_destroy();
                header("Location: index.php?pesan=transaksi_gagal");
                exit;
            }
        }
    }
}

        function requestStatistikTerbaru() {
            fetch('index.php?page=api_live_count')
                .then(response => {
                    if (!response.ok) throw new Error('Network error saat polling data.');
                    return response.json();
                })
                .then(data => {
                    // 1. Update Data Kartu Statistik Partisipasi
                    document.getElementById('stat-total-dpt').innerText = data.partisipasi.total_dpt;
                    document.getElementById('stat-sudah-memilih').innerText = data.partisipasi.sudah_memilih;
                    document.getElementById('stat-belum-memilih').innerText = data.partisipasi.belum_memilih;
                    document.getElementById('stat-persen-kehadiran').innerText = data.partisipasi.persentase_kehadiran + '%';
                    document.getElementById('stat-timestamp').innerText = 'Sinkronisasi Terakhir: ' + data.timestamp;

                    // 2. Update Persentase & Lebar Progress Bar Masing-masing Paslon
                    data.kandidat.forEach(k => {
                        const labelText = document.getElementById('label-persen-' + k.id_kandidat);
                        const progressBar = document.getElementById('bar-' + k.id_kandidat);

                        if (labelText && progressBar) {
                            labelText.innerText = `${k.persentase}% (${k.suara} Suara)`;
                            progressBar.style.width = k.persentase + '%';
                        }
                    });
                })
                .catch(err => {
                    console.warn('Gagal menarik data sinkronisasi live: ', err);
                });
        }

        // Jalankan polling setiap 2500 milidetik (2.5 detik)
        setInterval(requestStatistikTerbaru, 2500);
    </script>
