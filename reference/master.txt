Inti dari semua aplikasi web itu cuma satu: Tabel Master (CRUD) + Tabel User (Login). Kalau soalnya ganti jadi "Pegawai", lu tinggal find & replace kata barang jadi pegawai. Logika kodenya 100% sama!

Biar paham luar dalam, kita bedah dulu Alur Super Komplitnya:

Database: Tempat lu nyiapin "lemari" (tabel) dan "laci" (kolom).

Model (Kuli Panggul): Kodenya cuma bertugas ngomong ke database (pake perintah SELECT, INSERT, UPDATE, DELETE).

Controller (Otak/Mandor): Dia yang nerima inputan dari form, nyuruh Model kerja, terus nentuin habis ini user mau dilempar ke halaman mana. Di sini logika waktu, session, dan pengecekan error ditaruh.

View (Wajah Aplikasi): Murni cuma kode HTML, Bootstrap, dan sedikit PHP buat nge-cetak data (pake echo). Nggak boleh ada logika database di sini.

--- A. models/UserModel.php ---
<?php
class UserModel {
    private $db;
    public function __construct($koneksi) { $this->db =$koneksi; }

    // Berlaku untuk Login aja MAUPUN pakai Register
    public function login($username, $password) {$username = mysqli_real_escape_string($this->db,$username);
        // Note: Password di sini nggak di md5() biar gampang ngetes ujian, kalau mau aman tambahin md5()
        $hasil =$this->db->query("SELECT * FROM users WHERE username = '$username' AND password = '$password'");
        return $hasil->fetch_assoc();
    }

    // KHUSUS kalau ada Form Register
    public function register($nama, $username,$password) {
        $nama = mysqli_real_escape_string($this->db, $nama);$username = mysqli_real_escape_string($this->db,$username);
        return $this->db->query("INSERT INTO users (nama_lengkap, username, password) VALUES ('$nama', '$username', '$password')");
    }
}
?>

<?php
require_once 'models/UserModel.php';

        if(session_status() === PHP_SESSION_NONE) session_start();
    }

        if($_SERVER['REQUEST_METHOD'] == 'POST') {

            if($user) {$_SESSION['id_user'] = $user['id_user'];$_SESSION['nama_lengkap'] = $user['nama_lengkap'];$_SESSION['sapaan_baru'] = true; // Triger Alert Sapaan di Dashboard
                header("Location: index.php?page=dashboard"); exit;

            header("Location: index.php?alert=gagal"); exit;

        if($_SERVER['REQUEST_METHOD'] == 'POST') {

        }
        require_once 'views/register.php';
    }

    public function logout() {
        session_destroy(); header("Location: index.php"); exit;
    }

--- A. models/MasterModel.php ---
<?php
class MasterModel {
    private $db;
    public function __construct($koneksi) { $this->db =$koneksi; }

    public function getAll() { return $this->db->query("SELECT * FROM master_data ORDER BY id_data DESC"); }
    public function getById($id) { return$this->db->query("SELECT * FROM master_data WHERE id_data = ".(int)$id)->fetch_assoc(); }

    public function tambah($nama, $jenis,$merk, $tahun,$harga, $kategori) {$nama = mysqli_real_escape_string($this->db,$nama);
        return $this->db->query("INSERT INTO master_data (nama_item, jenis, merk, tahun_keluar, harga, kategori) 
                                 VALUES ('$nama', '$jenis', '$merk', '$tahun', '$harga', '$kategori')");
    }

    public function update($id, $nama,$jenis, $merk,$tahun, $harga,$kategori) {
        $id = (int)$id;
        $nama = mysqli_real_escape_string($this->db,$nama);
        return $this->db->query("UPDATE master_data SET nama_item='$nama', jenis='$jenis', merk='$merk', 
                                 tahun_keluar='$tahun', harga='$harga', kategori='$kategori' WHERE id_data=$id");
    }

    public function hapus($id) { return$this->db->query("DELETE FROM master_data WHERE id_data = ".(int)$id); }
}
?>

        if($_SERVER['REQUEST_METHOD'] == 'POST') {$this->model->tambah($_POST['nama'],$_POST['jenis'], $_POST['merk'],$_POST['tahun'], $_POST['harga'],$_POST['kategori']);
            header("Location: index.php?page=data&pesan=tambah"); exit;
        }
        require_once 'views/master_tambah.php';
    }

    public function form_edit() {
        if($_SERVER['REQUEST_METHOD'] == 'POST') {
            $this->model->update($_POST['id'], $_POST['nama'],$_POST['jenis'], $_POST['merk'],$_POST['tahun'], $_POST['harga'],$_POST['kategori']);
            header("Location: index.php?page=data&pesan=edit"); exit;
        } else {
            $data_edit = $this->model->getById($_GET['id']);
            require_once 'views/master_edit.php';
        }
    }

    public function aksi_hapus() {
        if(isset($_GET['id'])) $this->model->hapus($_GET['id']);
        header("Location: index.php?page=data&pesan=hapus"); exit;
    }
}
?>

<?php

$auth = new AuthController($koneksi);$master = new MasterController($koneksi);$page = isset($_GET['page']) ?$_GET['page'] : '';

switch ($page) {
    // ROUTE LOGIN & REGISTER
    case 'register': $auth->register(); break;
    case 'logout': $auth->logout(); break;

    // ROUTE DASHBOARD (Aman, dicek belum login ga bisa masuk)
    case 'dashboard': 
        if(!isset($_SESSION['id_user'])) { header("Location: index.php"); exit; }
        require_once 'views/dashboard.php'; 
        break;

    // ROUTE CRUD DATA (Semua diarahkan ke MasterController)
    case 'data': 
        if(!isset($_SESSION['id_user'])) { header("Location: index.php"); exit; }
        $master->index(); 
        break;
    case 'tambah_data': 
        if(!isset($_SESSION['id_user'])) { header("Location: index.php"); exit; }
        $master->form_tambah(); 
        break;
    case 'edit_data': 
        if(!isset($_SESSION['id_user'])) { header("Location: index.php"); exit; }
        $master->form_edit(); 
        break;
    case 'hapus_data': 
        if(!isset($_SESSION['id_user'])) { header("Location: index.php"); exit; }
        $master->aksi_hapus(); 
        break;

    // DEFAULT ROUTE -> Ke halaman LOGIN
    default:
        if(isset($_SESSION['id_user'])) { header("Location: index.php?page=dashboard"); exit; }
        $auth->login(); 
        break;
}
