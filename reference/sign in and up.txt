1. DATABASE (Buat di phpMyAdmin)
====================================================================
Buat database bernama: `db_sekolah`

-- Tabel untuk Login & Register
CREATE TABLE users (
    id_user INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    nama_lengkap VARCHAR(100) NOT NULL
);

-- Tabel untuk CRUD Buku
CREATE TABLE buku (
    id_buku INT AUTO_INCREMENT PRIMARY KEY,
    judul VARCHAR(150) NOT NULL,
    penulis VARCHAR(100) NOT NULL,
    kategori VARCHAR(50) NOT NULL,
    tahun_terbit INT(4) NOT NULL
);

--- A. models/UserModel.php (Tambah Fitur Cek User & Register) ---
<?php
class UserModel {
    private $db;
    public function __construct($koneksi) { $this->db =$koneksi; }

    public function cekLogin($username,$password) {
        $username = mysqli_real_escape_string($this->db, $username);$password = mysqli_real_escape_string($this->db,$password);
        $hasil =$this->db->query("SELECT * FROM users WHERE username = '$username' AND password = '$password'");
        return ($hasil->num_rows > 0) ?$hasil->fetch_assoc() : false;
    }

    public function cekUsernameAda($username) {$username = mysqli_real_escape_string($this->db,$username);
        $hasil =$this->db->query("SELECT * FROM users WHERE username = '$username'");
        return $hasil->num_rows > 0;
    }

    public function register($username,$password, $nama) {$username = mysqli_real_escape_string($this->db,$username);
        $password = mysqli_real_escape_string($this->db, $password);$nama = mysqli_real_escape_string($this->db,$nama);
        return $this->db->query("INSERT INTO users (username, password, nama_lengkap) VALUES ('$username', '$password', '$nama')");
    }
}
?>

--- B. models/BukuModel.php (CRUD Penuh) ---
<?php
class BukuModel {
    private $db;
    public function __construct($koneksi) { $this->db =$koneksi; }

    public function getSemuaBuku() { return $this->db->query("SELECT * FROM buku ORDER BY id_buku DESC"); }
    public function getBukuById($id) { return$this->db->query("SELECT * FROM buku WHERE id_buku = ".(int)$id)->fetch_assoc(); }

    public function tambahBuku($judul,$penulis, $kategori,$tahun) {
        $judul = mysqli_real_escape_string($this->db, $judul);$penulis = mysqli_real_escape_string($this->db,$penulis);
        return $this->db->query("INSERT INTO buku (judul, penulis, kategori, tahun_terbit) VALUES ('$judul', '$penulis', '$kategori', '$tahun')");
    }

    public function updateBuku($id, $judul,$penulis, $kategori,$tahun) {
        $id = (int)$id;
        $judul = mysqli_real_escape_string($this->db, $judul);$penulis = mysqli_real_escape_string($this->db,$penulis);
        return $this->db->query("UPDATE buku SET judul='$judul', penulis='$penulis', kategori='$kategori', tahun_terbit='$tahun' WHERE id_buku=$id");
    }

    public function hapusBuku($id) { return$this->db->query("DELETE FROM buku WHERE id_buku = ".(int)$id); }
}
?>

--- A. controllers/AuthController.php ---
<?php
require_once 'models/UserModel.php';
class AuthController {
    private $model;
    public function __construct($koneksi) { 
        $this->model = new UserModel($koneksi); 
        if (session_status() === PHP_SESSION_NONE) session_start();
    }

    public function login() {
        if(isset($_SESSION['is_login'])) { header("Location: index.php?page=dashboard"); exit; }
        if ($_SERVER['REQUEST_METHOD'] == 'POST') {
            $user =$_POST['username'];
            $pass =$_POST['password'];
            $cek =$this->model->cekLogin($user,$pass);
            if ($cek) {$_SESSION['is_login'] = true;
                $_SESSION['nama_lengkap'] = $cek['nama_lengkap'];$_SESSION['welcome_alert'] = true; // TRIGER POPUP NYAPA DI DASHBOARD NANTI
                header("Location: index.php?page=dashboard"); exit;
            } else {
                header("Location: index.php?error=1"); exit;
            }
        }
        require_once 'views/login.php';
    }

    public function register() {
        if ($_SERVER['REQUEST_METHOD'] == 'POST') {
            $user =$_POST['username'];
            $pass =$_POST['password'];
            $nama =$_POST['nama_lengkap'];
            
            if ($this->model->cekUsernameAda($user)) {
                header("Location: index.php?page=register&error=ada"); exit;
            } else {
                $this->model->register($user, $pass,$nama);
                header("Location: index.php?sukses=1"); exit;
            }
        }
        require_once 'views/register.php';
    }

    public function logout() { session_destroy(); header("Location: index.php"); exit; }
}
?>

--- B. controllers/BukuController.php ---
<?php
require_once 'models/BukuModel.php';
class BukuController {
    private $model;
    public function __construct($koneksi) { $this->model = new BukuModel($koneksi); }

    public function index() {
        $data_buku =$this->model->getSemuaBuku();
        require_once 'views/buku_tampil.php';
    }

    public function tambah() {
        if ($_SERVER['REQUEST_METHOD'] == 'POST') {$this->model->tambahBuku($_POST['judul'],$_POST['penulis'], $_POST['kategori'],$_POST['tahun']);
            header("Location: index.php?page=buku&pesan=sukses_tambah"); exit;
        }
        require_once 'views/buku_tambah.php';
    }

    public function edit() {
        if ($_SERVER['REQUEST_METHOD'] == 'POST') {
            $this->model->updateBuku($_POST['id_buku'], $_POST['judul'],$_POST['penulis'], $_POST['kategori'],$_POST['tahun']);
            header("Location: index.php?page=buku&pesan=sukses_edit"); exit;
        } else {
            $buku_edit = $this->model->getBukuById($_GET['id']);
            require_once 'views/buku_edit.php';
        }
    }

    public function hapus() {
        if (isset($_GET['id'])) $this->model->hapusBuku($_GET['id']);
        header("Location: index.php?page=buku&pesan=sukses_hapus"); exit;
    }
}
?>

6. ROUTER (index.php) - Taruh Paling Luar!
====================================================================
<?php
require_once 'config/database.php';
require_once 'controllers/AuthController.php';
require_once 'controllers/BukuController.php';

$auth = new AuthController($koneksi);
$buku = new BukuController($koneksi);

$page = isset($_GET['page']) ?$_GET['page'] : '';

switch ($page) {
    // Rute Login & Register
    case 'register': $auth->register(); break;
    case 'logout': $auth->logout(); break;
    
    // Rute Dashboard
    case 'dashboard': require_once 'views/dashboard.php'; break;
    
    // Rute CRUD Buku
    case 'buku': $buku->index(); break;
    case 'tambah_buku': $buku->tambah(); break;
    case 'edit_buku': $buku->edit(); break;
    case 'hapus_buku': $buku->hapus(); break;
    
    // Default masuk ke Login
    default: $auth->login(); break;
}
?>
