LANGKAH 1: DATABASE & MAGIC TRIGGER (phpMyAdmin)
====================================================================
Buat database bernama `db_tabungan_sekolah`.

-- 1. Tabel Users (Untuk Admin/Teller)
CREATE TABLE users (
    id_user INT AUTO_INCREMENT PRIMARY KEY,
    nama_lengkap VARCHAR(100) NOT NULL,
    username VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL
);
INSERT INTO users (nama_lengkap, username, password) VALUES ('Teller Utama', 'admin', '12345');

-- 2. Tabel Master Siswa (Menyimpan Saldo Akhir)
CREATE TABLE siswa (
    id_siswa INT AUTO_INCREMENT PRIMARY KEY,
    nis VARCHAR(20) NOT NULL,
    nama_siswa VARCHAR(100) NOT NULL,
    kelas VARCHAR(50) NOT NULL,
    saldo_akhir INT(11) DEFAULT 0
);
-- Masukin Dummy Siswa
INSERT INTO siswa (nis, nama_siswa, kelas, saldo_akhir) VALUES 
('101', 'Rixsan Joulfiand', 'X RPL 1', 0),
('102', 'Budi Santoso', 'X RPL 2', 0);

-- 3. Tabel Riwayat Transaksi
CREATE TABLE transaksi (
    id_transaksi INT AUTO_INCREMENT PRIMARY KEY,
    id_siswa INT NOT NULL,
    tgl_transaksi DATETIME NOT NULL,
    jenis ENUM('Setor', 'Tarik') NOT NULL,
    nominal INT(11) NOT NULL,
    keterangan TEXT NOT NULL
);

--------------------------------------------------------------------
-- KODE SAKTI: MYSQL TRIGGER (Jalankan kode ini di tab SQL phpMyAdmin)
-- Fungsi: Otomatis nambah/ngurangin saldo_akhir saat ada transaksi masuk!
--------------------------------------------------------------------
DELIMITER $$ CREATE TRIGGER otomatis_update_saldo  AFTER INSERT ON transaksi FOR EACH ROW  BEGIN     IF NEW.jenis = 'Setor' THEN         UPDATE siswa SET saldo_akhir = saldo_akhir + NEW.nominal WHERE id_siswa = NEW.id_siswa;     ELSEIF NEW.jenis = 'Tarik' THEN         UPDATE siswa SET saldo_akhir = saldo_akhir - NEW.nominal WHERE id_siswa = NEW.id_siswa;     END IF; END$$
DELIMITER ;
--------------------------------------------------------------------

--- C. models/TabunganModel.php ---
<?php
class TabunganModel {
    private $db;
    public function __construct($koneksi) { $this->db =$koneksi; }

    // Ambil daftar siswa untuk Dropdown di form
    public function getSemuaSiswa() {
        return $this->db->query("SELECT * FROM siswa ORDER BY nama_siswa ASC");
    }

    // Ambil saldo spesifik siswa (Buat validasi Tarik Tunai)
    public function getSaldoSiswa($id_siswa) {
        $id = (int)$id_siswa;
        $hasil =$this->db->query("SELECT saldo_akhir FROM siswa WHERE id_siswa = $id");
        $data =$hasil->fetch_assoc();
        return $data['saldo_akhir'];
    }

    // Ambil riwayat transaksi lengkap pakai INNER JOIN
    public function getRiwayatTransaksi() {
        $query = "SELECT t.*, s.nama_siswa, s.kelas, s.saldo_akhir 
                  FROM transaksi t 
                  JOIN siswa s ON t.id_siswa = s.id_siswa 
                  ORDER BY t.tgl_transaksi DESC";
        return $this->db->query($query);
    }

    // Hitung Total Uang di Bank (Untuk Dashboard)
    public function getTotalUangBank() {
        $hasil =$this->db->query("SELECT SUM(saldo_akhir) as total FROM siswa");
        return $hasil->fetch_assoc()['total'];
    }

    // Fungsi Insert Transaksi (Otomatis memicu Trigger di Database)
    public function tambahTransaksi($id_siswa,$jenis, $nominal,$ket) {
        $id_siswa = (int)$id_siswa;
        $nominal = (int)$nominal;
        $tgl = date('Y-m-d H:i:s'); // Rekam jam persis saat eksekusi$ket = mysqli_real_escape_string($this->db,$ket);
        
        return $this->db->query("INSERT INTO transaksi (id_siswa, tgl_transaksi, jenis, nominal, keterangan) 
                                 VALUES ($id_siswa, '$tgl', '$jenis', $nominal, '$ket')");
    }
}
?>

--- A. controllers/AuthController.php ---
<?php
require_once 'models/UserModel.php';
class AuthController {
    private $model;
    public function __construct($koneksi) { 
        $this->model = new UserModel($koneksi); 
        if(session_status() === PHP_SESSION_NONE) session_start();
    }
    public function login() {
        if(isset($_SESSION['id_user'])) { header("Location: index.php?page=dashboard"); exit; }
        if($_SERVER['REQUEST_METHOD'] == 'POST') {
            $user =$this->model->login($_POST['username'],$_POST['password']);
            if($user) {
                $_SESSION['id_user'] =$user['id_user'];
                $_SESSION['nama_lengkap'] =$user['nama_lengkap'];
                header("Location: index.php?page=dashboard&alert=welcome"); exit;
            }
            header("Location: index.php?alert=gagal"); exit;
        }
        require_once 'views/login.php';
    }
    public function logout() { session_destroy(); header("Location: index.php"); exit; }
}
?>

--- B. controllers/TabunganController.php ---
<?php
require_once 'models/TabunganModel.php';
class TabunganController {
    private $model;
    public function __construct($koneksi) { $this->model = new TabunganModel($koneksi); }

    public function dashboard() {
        $total_uang =$this->model->getTotalUangBank();
        require_once 'views/dashboard.php';
    }

    public function riwayat() {
        $data_transaksi =$this->model->getRiwayatTransaksi();
        require_once 'views/transaksi_tampil.php';
    }

    public function setor_tarik() {
        $data_siswa =$this->model->getSemuaSiswa();

        if($_SERVER['REQUEST_METHOD'] == 'POST') {
            $id_siswa =$_POST['id_siswa'];
            $jenis =$_POST['jenis'];
            $nominal =$_POST['nominal'];
            $keterangan =$_POST['keterangan'];

            // LOGIKA PENTING: Jika Tarik Tunai, cek dulu saldo cukup apa nggak!
            if($jenis == 'Tarik') {$saldo_saat_ini = $this->model->getSaldoSiswa($id_siswa);
                if($nominal >$saldo_saat_ini) {
                    header("Location: index.php?page=tambah_transaksi&pesan=saldo_kurang"); 
                    exit;
                }
            }

            // Jika saldo cukup (atau jika jenisnya Setor), eksekusi simpan!
            $this->model->tambahTransaksi($id_siswa,$jenis, $nominal,$keterangan);
            header("Location: index.php?page=riwayat&pesan=sukses_transaksi");
            exit;
        }
        require_once 'views/transaksi_tambah.php';
    }
}
?>

Penjelasan Kunci (Kenapa Sistem Ini Mahal Harganya):
Trigger MySQL: Saat lu nge-klik tombol "ProSES TRANSAKSI", controller cuma nyuruh database nyimpan data transaksi. Ajaibnya, di detik yang sama, MySQL langsung sadar "Eh ada transaksi nih!" trus dia baca DELIMITER yang udah lu pasang. Kalau jenisnya 'Setor', kolom saldo_akhir milik siswa yang bersangkutan langsung ditambah nominal. Sistem PHP lu jadi enteng banget kerjanya!

Validasi Tarik Tunai: Di TabunganController->setor_tarik(), ada logika pengecekan maut:
if($nominal > $saldo_saat_ini) { tolak! }
Ini mencegah yang namanya "Saldo Minus".

Format Rupiah Dinamis: Tabel dan Dropdown di View lu udah otomatis dikasih format titik ribuan khas Indonesia pakai fungsi number_format(), bikin tampilannya kerasa kayak ATM beneran.
