Deskripsi: Sistem ini menampilkan daftar siswa dengan kolom tanggal 1 sampai 30 ke samping. Guru bisa mengubah status (H/I/S/A) langsung dari tabel dan menyimpan semuanya sekaligus dengan satu tombol.

--- 1. DATABASE (phpMyAdmin) ---
Jalankan kode SQL ini untuk membuat tabel siswa, tabel kehadiran, dan MENGUNCI data agar tidak ganda (duplikat) di tanggal yang sama.

-- Tabel data siswa
CREATE TABLE siswa (
    id_siswa INT AUTO_INCREMENT PRIMARY KEY,
    nama_siswa VARCHAR(100) NOT NULL
);

-- Tabel data kehadiran
CREATE TABLE kehadiran (
    id_hadir INT AUTO_INCREMENT PRIMARY KEY,
    id_siswa INT NOT NULL,
    tanggal DATE NOT NULL,
    status ENUM('H', 'I', 'S', 'A') NOT NULL
);

-- KUNCI PENTING: Mencegah 1 siswa punya 2 absen di hari yang sama
ALTER TABLE kehadiran ADD UNIQUE KEY unique_absen (id_siswa, tanggal);

--- 2. MODEL (models/AbsensiModel.php) ---
Fungsi: Membentuk array dari tanggal 1-30 (Pivot) dan fungsi canggih ON DUPLICATE KEY UPDATE untuk menimpa data yang sudah ada.

<?php
class AbsensiModel {
    private $db;

    public function __construct($koneksi) {
        $this->db = $koneksi;
    }

    public function getRekapBulanan() {
        $rekap = [];
        
        // 1. Siapkan kerangka array 1-30 untuk semua siswa
        $query_siswa = $this->db->query("SELECT * FROM siswa ORDER BY id_siswa ASC");
        while($s = $query_siswa->fetch_assoc()) {
            $id = $s['id_siswa'];
            $rekap[$id] = [
                'nama' => $s['nama_siswa'],
                'absen' => array_fill(1, 30, '-') 
            ];
        }

        // 2. Ambil absen bulan ini, lalu timpa tanda strip (-) dengan H/I/S/A
        $query_absen = $this->db->query("SELECT id_siswa, DAY(tanggal) as tgl, status FROM kehadiran WHERE MONTH(tanggal) = MONTH(CURDATE()) AND YEAR(tanggal) = YEAR(CURDATE())");
        while($a = $query_absen->fetch_assoc()) {
            $id = $a['id_siswa'];
            $tgl = (int)$a['tgl'];
            if(isset($rekap[$id]) && $tgl <= 30) {
                $rekap[$id]['absen'][$tgl] = $a['status'];
            }
        }
        return $rekap;
    }

    // Fungsi Insert / Update dalam 1 perintah SQL
    public function simpanAtauUpdateAbsen($id_siswa, $tanggal, $status) {
        $id_siswa = (int)$id_siswa;
        $query = "INSERT INTO kehadiran (id_siswa, tanggal, status) 
                  VALUES ($id_siswa, '$tanggal', '$status') 
                  ON DUPLICATE KEY UPDATE status = '$status'";
        return $this->db->query($query);
    }

    public function hapusAbsenMatrix($id_siswa, $tanggal) {
        $id_siswa = (int)$id_siswa;
        return $this->db->query("DELETE FROM kehadiran WHERE id_siswa = $id_siswa AND tanggal = '$tanggal'");
    }
}
?>

--- 3. CONTROLLER (controllers/AbsensiController.php) ---
Fungsi: Menangkap ratusan data dari form (array multi-dimensi) lalu memecahnya pakai foreach.

<?php
require_once 'models/AbsensiModel.php';

class AbsensiController {
    private $model;

    public function __construct($koneksi) {
        $this->model = new AbsensiModel($koneksi);
    }

    public function index() {
        $data_rekap = $this->model->getRekapBulanan();
        require_once 'views/rekap_absensi.php';
    }

    public function simpanMatrix() {
        if ($_SERVER['REQUEST_METHOD'] == 'POST') {
            $bulan_ini = date('Y-m'); // Format YYYY-MM
            $data_absen = $_POST['absen'];
            
            foreach ($data_absen as $id_siswa => $tanggal_data) {
                foreach ($tanggal_data as $tgl => $status) {
                    // Gabungkan tahun, bulan, dan tanggal
                    $tgl_full = $bulan_ini . '-' . str_pad($tgl, 2, '0', STR_PAD_LEFT);
                    
                    if ($status == '-') {
                        $this->model->hapusAbsenMatrix($id_siswa, $tgl_full);
                    } else {
                        $this->model->simpanAtauUpdateAbsen($id_siswa, $tgl_full, $status);
                    }
                }
            }
            header("Location: index.php?page=absensi&pesan=sukses_simpan");
            exit;
        }
    }
}
?>

--- 5. ROUTING (index.php) ---
Fungsi: Memastikan url ?page=simpan_matrix mengarah ke fungsi yang tepat di Controller.

<?php
require_once 'config/database.php';
require_once 'controllers/AbsensiController.php';

$absensi = new AbsensiController($koneksi);
$page = isset($_GET['page']) ? $_GET['page'] : 'dashboard';

switch ($page) {
    case 'dashboard':
        require_once 'views/dashboard.php';
        break;
        
    case 'absensi':
        $absensi->index(); 
        break;

    case 'simpan_matrix': // <--- RUTE BARU UNTUK SAVE MASSAL
        $absensi->simpanMatrix();
        break;

    default:
        require_once 'views/dashboard.php';
        break;
}
