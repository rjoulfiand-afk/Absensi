--- models/KalkulatorModel.php ---
<?php
class KalkulatorModel {
    private $db;
    public function __construct($koneksi) { $this->db =$koneksi; }

    // READ
    public function getRiwayat() {
        return $this->db->query("SELECT * FROM riwayat_hitung ORDER BY id_riwayat DESC");
    }

    // CREATE (Simpan hasil hitungan ke DB)
    public function simpanHitung($a1,$op, $a2,$hasil) {
        return $this->db->query("INSERT INTO riwayat_hitung (angka_1, operator, angka_2, hasil) 
                                 VALUES ('$a1', '$op', '$a2', '$hasil')");
    }

    // UPDATE (Kasih catatan buat riwayat itungan)
    public function updateCatatan($id,$catatan) {
        $id = (int)$id;
        $catatan = mysqli_real_escape_string($this->db,$catatan);
        return $this->db->query("UPDATE riwayat_hitung SET catatan = '$catatan' WHERE id_riwayat =$id");
    }

    // DELETE
    public function hapusRiwayat($id) {
        return $this->db->query("DELETE FROM riwayat_hitung WHERE id_riwayat = ".(int)$id);
    }
    
    // DELETE ALL (Bersihin layar)
    public function bersihkanSemua() {
        return $this->db->query("TRUNCATE TABLE riwayat_hitung");
    }
}
?>

LANGKAH 3: CONTROLLER (Logika Matematikanya di sini!)
====================================================================

--- controllers/KalkulatorController.php ---
<?php
require_once 'models/KalkulatorModel.php';

class KalkulatorController {
    private $model;
    public function __construct($koneksi) { $this->model = new KalkulatorModel($koneksi); }

    public function index() {
        $riwayat =$this->model->getRiwayat();
        require_once 'views/kalkulator_view.php';
    }

    // LOGIKA KALKULATORNYA CUY
    public function prosesHitung() {
        if ($_SERVER['REQUEST_METHOD'] == 'POST') {$angka1 = (float)$_POST['angka1'];$operator = $_POST['operator'];$angka2 = (float)$_POST['angka2'];$hasil = 0;

            // Proses Matematika
            switch ($operator) {
                case '+': $hasil = $angka1 +$angka2; break;
                case '-': $hasil = $angka1 -$angka2; break;
                case 'x': $hasil = $angka1 * $angka2; break;
                case '/': 
                    // Pencegahan Error Dibagi Nol
                    if($angka2 == 0) {
                        header("Location: index.php?pesan=error_nol"); exit;
                    }
                    $hasil = $angka1 / $angka2; 
                    break;
            }

            // Langsung simpan ke Database (CREATE)
            $this->model->simpanHitung($angka1,$operator, $angka2,$hasil);
            header("Location: index.php?pesan=sukses_hitung"); exit;
        }
    }

    public function simpanCatatan() {
        if ($_SERVER['REQUEST_METHOD'] == 'POST') {$this->model->updateCatatan($_POST['id_riwayat'],$_POST['catatan']);
            header("Location: index.php?pesan=sukses_update"); exit;
        }
    }

    public function hapus() {
        if (isset($_GET['id'])) {
            $this->model->hapusRiwayat($_GET['id']);
            header("Location: index.php?pesan=sukses_hapus"); exit;
        }
    }
    
    public function reset() {
        $this->model->bersihkanSemua();
        header("Location: index.php?pesan=sukses_reset"); exit;
    }
}
?>

<?php

switch ($page) {
    case 'hitung':
        $app->prosesHitung(); break;
    case 'catatan':
        $app->simpanCatatan(); break;
    case 'hapus':
        $app->hapus(); break;
    case 'reset':
        $app->reset(); break;
    default:
        $app->index(); break;
}
