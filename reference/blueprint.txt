LANGKAH 1: PEMBUATAN DATABASE (phpMyAdmin)
====================================================================
Buat database bernama `db_perpus_advance`, lalu eksekusi SQL ini:

-- Tabel Users (Untuk Login Petugas/Admin)
CREATE TABLE users (
    id_user INT AUTO_INCREMENT PRIMARY KEY,
    nama_lengkap VARCHAR(100) NOT NULL,
    username VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL
);

-- Tabel Peminjaman (Inti Aplikasi)
CREATE TABLE peminjaman (
    id_pinjam INT AUTO_INCREMENT PRIMARY KEY,
    nama_peminjam VARCHAR(100) NOT NULL,
    kelas VARCHAR(50) NOT NULL,
    buku_dipinjam VARCHAR(150) NOT NULL,
    waktu_pinjam DATETIME NOT NULL,
    status ENUM('Dipinjam', 'Dikembalikan') DEFAULT 'Dipinjam'
);

--- B. models/UserModel.php ---
<?php
class UserModel {
    private $db;
    public function __construct($koneksi) { $this->db =$koneksi; }

    public function register($nama, $username,$password) {
        $nama = mysqli_real_escape_string($this->db, $nama);$username = mysqli_real_escape_string($this->db,$username);
        $password = md5($password); 
        return $this->db->query("INSERT INTO users (nama_lengkap, username, password) VALUES ('$nama', '$username', '$password')");
    }

    public function cekUsername($username) {$username = mysqli_real_escape_string($this->db,$username);
        return $this->db->query("SELECT * FROM users WHERE username = '$username'")->num_rows > 0;
    }

    public function login($username, $password) {$username = mysqli_real_escape_string($this->db,$username);
        $password = md5($password);
        $hasil =$this->db->query("SELECT * FROM users WHERE username = '$username' AND password = '$password'");
        return $hasil->fetch_assoc();
    }
}
?>

--- C. models/PinjamModel.php ---
<?php
class PinjamModel {
    private $db;
    public function __construct($koneksi) { $this->db =$koneksi; }

    // Tampil data, diurutkan dari yang paling baru minjam
    public function getSemuaPinjaman() {
        return $this->db->query("SELECT * FROM peminjaman ORDER BY waktu_pinjam DESC");
    }

    public function getPinjamanById($id) {
        $id = (int)$id;
        return $this->db->query("SELECT * FROM peminjaman WHERE id_pinjam = $id")->fetch_assoc();
    }

    // Waktu pinjam otomatis dilempar dari controller (Real-time server)
    public function tambahPinjaman($nama, $kelas,$buku, $waktu) {$nama = mysqli_real_escape_string($this->db,$nama);
        $kelas = mysqli_real_escape_string($this->db, $kelas);$buku = mysqli_real_escape_string($this->db,$buku);
        return $this->db->query("INSERT INTO peminjaman (nama_peminjam, kelas, buku_dipinjam, waktu_pinjam) 
                                 VALUES ('$nama', '$kelas', '$buku', '$waktu')");
    }

    public function updateStatus($id,$status) {
        $id = (int)$id;
        $status = mysqli_real_escape_string($this->db,$status);
        return $this->db->query("UPDATE peminjaman SET status = '$status' WHERE id_pinjam =$id");
    }

    public function hapusPinjaman($id) {
        $id = (int)$id;
        return $this->db->query("DELETE FROM peminjaman WHERE id_pinjam = $id");
    }
}
?>

--- A. controllers/AuthController.php ---
<?php
require_once 'models/UserModel.php';

class AuthController {
    private $model;
    public function __construct($koneksi) { 
        $this->model = new UserModel($koneksi); 
        if (session_status() === PHP_SESSION_NONE) session_start();
    }

    public function prosesLogin() {
        if ($_SERVER['REQUEST_METHOD'] == 'POST') {
            $user =$this->model->login($_POST['username'],$_POST['password']);
            if ($user) {
                $_SESSION['id_user'] =$user['id_user'];
                $_SESSION['nama_lengkap'] =$user['nama_lengkap'];
                header("Location: index.php?page=dashboard"); exit;
            }
            header("Location: index.php?alert=gagal"); exit;
        }
        require_once 'views/login.php';
    }

    public function prosesRegister() {
        if ($_SERVER['REQUEST_METHOD'] == 'POST') {
            if ($this->model->cekUsername($_POST['username'])) {
                header("Location: index.php?page=register&alert=ada"); exit;
            }
            $this->model->register($_POST['nama_lengkap'], $_POST['username'],$_POST['password']);
            header("Location: index.php?alert=sukses_daftar"); exit;
        }
        require_once 'views/register.php';
    }

    public function logout() {
        session_destroy(); header("Location: index.php"); exit;
    }
}
?>

--- B. controllers/PinjamController.php ---
<?php
require_once 'models/PinjamModel.php';

class PinjamController {
    private $model;
    public function __construct($koneksi) { 
        $this->model = new PinjamModel($koneksi); 
        // KUNCI: Kunci waktu server ke wilayah Indonesia (WIB)
        date_default_timezone_set('Asia/Jakarta');
    }

    public function index() {
        $data_pinjam =$this->model->getSemuaPinjaman();
        require_once 'views/pinjam_tampil.php';
    }

    public function tambah() {
        if ($_SERVER['REQUEST_METHOD'] == 'POST') {
            // Tangkap waktu secara presisi tepat saat tombol "Simpan" ditekan
            $waktu_sekarang = date('Y-m-d H:i:s');$this->model->tambahPinjaman($_POST['nama'],$_POST['kelas'], $_POST['buku'],$waktu_sekarang);
            header("Location: index.php?page=pinjam&pesan=sukses_tambah"); exit;
        }
        require_once 'views/pinjam_tambah.php';
    }

    public function edit() {
        if ($_SERVER['REQUEST_METHOD'] == 'POST') {$this->model->updateStatus($_POST['id_pinjam'],$_POST['status']);
            header("Location: index.php?page=pinjam&pesan=sukses_edit"); exit;
        } else {
            $data_edit = $this->model->getPinjamanById($_GET['id']);
            require_once 'views/pinjam_edit.php';
        }
    }

    public function hapus() {
        if(isset($_GET['id'])) $this->model->hapusPinjaman($_GET['id']);
        header("Location: index.php?page=pinjam&pesan=sukses_hapus"); exit;
    }
}
?>

LANGKAH 5: ROUTER UTAMA (index.php)
====================================================================
Taruh file ini di folder utama `project_perpus` lu.

<?php
session_start();
require_once 'config/database.php';
require_once 'controllers/AuthController.php';
require_once 'controllers/PinjamController.php';

$auth = new AuthController($koneksi);
$pinjam = new PinjamController($koneksi);

$page = isset($_GET['page']) ?$_GET['page'] : '';

switch ($page) {
    // AUTH ROUTE
    case 'register': $auth->prosesRegister(); break;
    case 'logout': $auth->logout(); break;

    // DASHBOARD ROUTE
    case 'dashboard':
        if (!isset($_SESSION['id_user'])) { header("Location: index.php"); exit; }
        require_once 'views/dashboard.php'; 
        break;

    // CRUD ROUTE
    case 'pinjam':
        if (!isset($_SESSION['id_user'])) { header("Location: index.php"); exit; }
        $pinjam->index(); 
        break;
    case 'tambah_pinjam':
        if (!isset($_SESSION['id_user'])) { header("Location: index.php"); exit; }
        $pinjam->tambah(); 
        break;
    case 'edit_pinjam':
        if (!isset($_SESSION['id_user'])) { header("Location: index.php"); exit; }
        $pinjam->edit(); 
        break;
    case 'hapus_pinjam':
        if (!isset($_SESSION['id_user'])) { header("Location: index.php"); exit; }
        $pinjam->hapus(); 
        break;

    // DEFAULT ROUTE
    default:
        if (isset($_SESSION['id_user'])) { header("Location: index.php?page=dashboard"); exit; }
        $auth->prosesLogin(); 
        break;
}
?>
